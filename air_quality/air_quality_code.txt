import time
import json
import ssl
import board
import busio
import adafruit_ads1x15.ads1115 as ADS
from adafruit_ads1x15.analog_in import AnalogIn
import paho.mqtt.client as mqtt

# ==========================================
# MQTT CONFIG
# ==========================================
MQTT_BROKER = "i5111178.ala.eu-central-1.emqxsl.com"
MQTT_PORT = 8883
MQTT_TOPIC = "smartcity/mq135"
MQTT_USER = "powerpuffgirls"
MQTT_PASS = "SCE2026grad"

client = mqtt.Client()

client.username_pw_set(
    MQTT_USER,
    MQTT_PASS
)

client.tls_set(cert_reqs=ssl.CERT_NONE)
client.tls_insecure_set(True)

client.connect(
    MQTT_BROKER,
    MQTT_PORT,
    60
)

# ==========================================
# ADS1115 SETUP
# ==========================================
i2c = busio.I2C(board.SCL, board.SDA)

ads = ADS.ADS1115(i2c)

mq2 = AnalogIn(ads, 2)
mq135 = AnalogIn(ads, 3)

print("Calculating Baseline...")
time.sleep(10)

# ==========================================
# FUNCTIONS
# ==========================================
def get_average(sensor, samples=20):

    total = 0

    for _ in range(samples):

        total += sensor.value

        time.sleep(0.05)

    return total / samples


baseline_mq2 = get_average(mq2)
baseline_mq135 = get_average(mq135)

mq2_buffer = []
mq135_buffer = []


def smooth(buffer, value):

    buffer.append(value)

    if len(buffer) > 5:
        buffer.pop(0)

    return sum(buffer) / len(buffer)


def percent_from_ratio(ratio):

    percent = (ratio - 1) * 100

    return max(0, min(percent, 100))


print("System Started...")

# ==========================================
# MAIN LOOP
# ==========================================
while True:

    try:

        # MQ2
        smoke_ratio = smooth(
            mq2_buffer,
            mq2.value / baseline_mq2
        )

        smoke_percent = percent_from_ratio(
            smoke_ratio
        )

        # MQ135
        gas_ratio = smooth(
            mq135_buffer,
            mq135.value / baseline_mq135
        )

        gas_percent = percent_from_ratio(
            gas_ratio
        )

        # AQI
        aqi = int(
            (smoke_percent + gas_percent) / 2
        )

        # CO
        co = round(
            smoke_percent / 10,
            2
        )

        # CO2
        co2 = int(
            400 + (gas_percent * 20)
        )

        # Smoke Status
        if smoke_percent < 10:
            smoke_status = "Low"
        elif smoke_percent < 40:
            smoke_status = "Medium"
        else:
            smoke_status = "High"

        # Benzene Status
        if gas_percent < 20:
            benzene_status = "Normal"
        elif gas_percent < 50:
            benzene_status = "Warning"
        else:
            benzene_status = "High"

        # Air Quality Level
        if aqi < 20:
            air_quality_level = "GOOD AIR"
        elif aqi < 50:
            air_quality_level = "MODERATE AIR"
        else:
            air_quality_level = "POLLUTED AIR"

        # System Status
        if smoke_percent > 40:
            system_status = "DANGER"
        elif gas_percent > 40:
            system_status = "WARNING"
        else:
            system_status = "SAFE"

        payload = {

            "device_id": "pi_device",

            "sensors": [

                {
                    "sensor_id": "mq2_sensor",
                    "type": "mq-2",
                    "co": co,
                    "smoke": smoke_status,
                    "power": 0.85
                },

                {
                    "sensor_id": "mq135_sensor",
                    "type": "mq-135",
                    "co2": co2,
                    "benzene": benzene_status,
                    "power": 0.60
                }

            ],

            "air_quality": {
                "aqi": aqi,
                "level": air_quality_level
            },

            "status": system_status,

            "power": 1.45

        }

        client.publish(
            MQTT_TOPIC,
            json.dumps(payload)
        )

        print(
            json.dumps(
                payload,
                indent=4
            )
        )

    except Exception as e:

        print("ERROR:", e)

    time.sleep(3)
