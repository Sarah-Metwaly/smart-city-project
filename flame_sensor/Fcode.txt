import time
import json
import paho.mqtt.client as mqtt
import board
import busio
import adafruit_ads1x15.ads1115 as ADS
from adafruit_ads1x15.analog_in import AnalogIn

# =========================
# MQTT
# =========================
client = mqtt.Client(mqtt.CallbackAPIVersion.VERSION2)
client.username_pw_set("powerpuffgirls", "SCE2026grad")
client.tls_set()
client.connect("i5111178.ala.eu-central-1.emqxsl.com", 8883, 60)
client.loop_start()

# =========================
# ADS1115 Setup
# =========================
i2c = busio.I2C(board.SCL, board.SDA)
ads = ADS.ADS1115(i2c)
flame_sensor = AnalogIn(ads, 1)

# =========================
# POWER MODEL (system-wide)
# =========================
BASE_POWER = 2.0
MAX_SENSOR_POWER = 3.0

# =========================
# 🔥 MORE SENSITIVE THRESHOLDS
# =========================
def process_flame(value):

    # 🔥 عالي الحساسية (بدري جدًا)
    if value < 12000:
        return "FLAME DETECTED", "DANGER", True, 1.0

    # 🔥 small flame أسرع في الاكتشاف
    elif value < 15000:
        return "SMALL FLAME", "WARNING", False, 0.6

    else:
        return "NO FLAME", "SAFE", False, 0.1


print("🔥 Flame Sensor Realtime System Running...")

try:
    while True:

        # 📡 realtime reading
        raw_value = flame_sensor.value

        status, risk, detected, power_factor = process_flame(raw_value)

        # ⚡ System Power (whole system)
        power = round(BASE_POWER + (MAX_SENSOR_POWER * power_factor), 2)

        #  JSON OUTPUT (مطابق للمطلوب)
        data = {
            "sensor_id": "flame_sensor",
            "status": status,
            "risk_level": risk,
            "is_flame_detected": detected,
            "power": power
        }

        # 🚀 publish
        client.publish("smartcity/flame", json.dumps(data))

        # 🖨 debug realtime
        print(f"RAW: {raw_value} | {data}")

        # ⚡ polling سريع جدًا (لحظي)
        time.sleep(0.2)

except KeyboardInterrupt:
    client.disconnect()
    print("Stopped safely")
